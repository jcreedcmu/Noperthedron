\chapter{Cayley coordinates}
\label{chap:cayley}

Angles make rotation matrices transcendental functions of the coordinates. Cayley
coordinates make them rational, so that every quantity in the case analysis becomes a
polynomial after clearing positive denominators.

\begin{definition}
\label{def:cayley}
\lean{Cayley.cayleyDenom, Cayley.cayleyNumeratorMatrix, Cayley.cayleyMatrix}
\leanok
For $r = (x, y, z) \in \R^3$ let $D(r) = 1 + x^2 + y^2 + z^2$ and
\[
  N(r) = \begin{pmatrix}
    1 + x^2 - y^2 - z^2 & 2(xy - z) & 2(xz + y) \\
    2(xy + z) & 1 - x^2 + y^2 - z^2 & 2(yz - x) \\
    2(xz - y) & 2(yz + x) & 1 - x^2 - y^2 + z^2
  \end{pmatrix}.
\]
The \emph{Cayley rotation} of $r$ is $\mathrm{Cay}(r) = N(r) / D(r)$. It is the
rotation by the unit quaternion proportional to $(1, x, y, z)$.
\end{definition}

The numerator $N(r)$ is a polynomial with integer coefficients, defined over any
commutative ring, and the denominator is positive.

Geometrically, if $a$ is a unit vector and $-\pi < \theta < \pi$, then
$r = \tan(\theta/2)\,a$ gives the rotation through angle $\theta$ about $a$.
In particular, $r = 0$ gives the identity, and small $r$ describes a small rotation.
A single Cayley chart misses rotations through angle $\pi$; the chart bases in
Definition~\ref{def:charts} allow the reduction to cover all relative rotations
using bounded Cayley coordinates.

\begin{lemma}
\label{lem:cayley_so3}
\lean{Cayley.cayleyMatrix_mem_SO3, Cayley.cayleyDenom_pos}
\leanok
\uses{def:cayley}
$D(r) > 0$ and $\mathrm{Cay}(r) \in SO(3)$ for every $r \in \R^3$.
\end{lemma}
\begin{proof}
\leanok
Direct computation: $N(r) N(r)^T = D(r)^2 I$ and $\det N(r) = D(r)^3$.
\end{proof}

\begin{lemma}[Rodrigues form]
\label{lem:cayley_rodrigues}
\lean{Cayley.num_sub_denom}
\leanok
\uses{def:cayley}
For every $S \in \R^3$,
\[
  N(r) S - D(r) S = 2\,\bigl(r \times S + r \times (r \times S)\bigr).
\]
Equivalently $\mathrm{Cay}(r) S - S = \frac{2}{D(r)} \bigl(r \times S + r \times (r \times S)\bigr)$.
\end{lemma}
\begin{proof}
\leanok
Expand $N(r) S = (1 - |r|^2) S + 2 r (r \cdot S) + 2 r \times S$ and use
$r \times (r \times S) = r (r \cdot S) - |r|^2 S$.
\end{proof}

The Rodrigues form is the source of the blow-up identity
(Lemma~\ref{lem:blowup_identity}): the difference $\mathrm{Cay}(r) S - S$ is divisible
by $r$.

\begin{lemma}
\label{lem:cayley_trace}
\lean{Cayley.trace_cayleyMatrix}
\leanok
\uses{def:cayley}
$\tr \mathrm{Cay}(r) = (3 - |r|^2) / (1 + |r|^2)$.
\end{lemma}
\begin{proof}
\leanok
Direct computation.
\end{proof}

\begin{theorem}
\label{thm:cayley_surj}
\lean{Cayley.exists_cayleyMatrix_of_trace_nonneg}
\leanok
\uses{lem:cayley_so3}
Every $A \in SO(3)$ with $\tr A \ge 0$ is $\mathrm{Cay}(r)$ for some $r$ with
$|r|^2 \le 3$.
\end{theorem}
\begin{proof}
\leanok
Let $d = 1 + \tr A > 0$ and let $r = (A_{21} - A_{12},\ A_{02} - A_{20},\ A_{10} - A_{01}) / d$
be the skew part of $A$ divided by $d$. (In terms of the unit quaternion $(q_0, q)$ of $A$,
$d = 4 q_0^2$ and $r = q / q_0$.) A rotation satisfies $A A^T = A^T A = I$ and is its own
cofactor matrix. These quadratic relations give, by linear combination,
\[
  r_0^2 = \frac{1 + A_{00} - A_{11} - A_{22}}{d}, \qquad
  r_0 r_1 = \frac{A_{01} + A_{10}}{d},
\]
and the four similar identities for the other squares and products. Hence
$|r|^2 = (3 - \tr A)/d \le 3$ and $D(r) = 4/d$, and substituting the six identities into
the entries of $N(r)/D(r)$ gives $A$.
\end{proof}

\section{Outer views}

\begin{definition}
\label{def:outer_rot}
\lean{Cayley.outerRot}
\leanok
\uses{def:cayley}
For $(u, v) \in \R^2$ let $s(u, v) = \mathrm{Cay}(v, -u, 0)$.
\end{definition}

\begin{lemma}
\label{lem:outer_view}
\lean{Cayley.outerRot_row2}
\leanok
\uses{def:outer_rot}
The third row of $s(u, v)$, the \emph{view} $s(u,v)^T e_z$, is the inverse
stereographic image
\[
  \frac{(2u,\ 2v,\ 1 - u^2 - v^2)}{1 + u^2 + v^2}.
\]
\end{lemma}
\begin{proof}
\leanok
Direct computation.
\end{proof}

Thus $(u,v)$ records the direction from which the outer polyhedron is viewed:
$(0,0)$ gives the positive $z$-axis, and $u^2+v^2=1$ gives horizontal views.
The rotation $s(u,v)$ chooses a particular orientation of the shadow in the screen;
any other rotation with the same view differs only by a rotation of the screen
(Lemma~\ref{lem:outer_unique}).
